1.CONFIGURE THE PHYSICAL INTERFACE
interface GigabitEthernet 0/0
 description LINK-TO-SWITCH-S1-TRUNK
 no shutdown                 ! <--- THIS IS THE MAGIC COMMAND THAT TURNS THE LIGHT GREEN!
 exit

2.CREATE SUBINTERFACES FOR EACH VLAN (Router-on-a-Stick)
VLAN 10: IT Department
interface GigabitEthernet 0/0.10
 description IT-DEPT-GATEWAY
 encapsulation dot1Q 10      ! Tells the router to tag/untag traffic for VLAN 10
 ip address 192.168.10.1 255.255.255.0
 exit

VLAN 20: Finance Department
interface GigabitEthernet 0/0.20
 description FINANCE-DEPT-GATEWAY
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 exit

VLAN 30: HR Department
interface GigabitEthernet 0/0.30
 description HR-DEPT-GATEWAY
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 exit

VLAN 99: Native/Management VLAN
interface GigabitEthernet 0/0.99
 description NATIVE-VLAN-MGMT
 encapsulation dot1Q 99 native  ! The 'native' keyword is required for the native VLAN subinterface
 ip address 192.168.99.1 255.255.255.0
 exit

3.SAVE THE CONFIGURATION
end
